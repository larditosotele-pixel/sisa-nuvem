function imprimir(orientacao) { const cards = Array.from(document.querySelectorAll('.grid.card')); if (cards.length === 0) { alert('Nenhum card encontrado na página!'); return; } // REGRA FIXA: 35 por página pra caber 100 em 3 folhas const colunas = orientacao === 'retrato'? 5 : 7; const linhas = orientacao === 'retrato'? 7 : 5; const cardsPorPagina = 35; // TRAVADO EM 35 // Calcula quantas páginas vai precisar const totalPaginas = Math.ceil(cards.length / cardsPorPagina); let html = ` Carômetro - Impressão `; for (let p = 0; p < totalPaginas; p++) { const inicio = p * cardsPorPagina + 1; const fim = Math.min((p + 1) * cardsPorPagina, cards.length); const cardsPagina = cards.slice(p * cardsPorPagina, (p + 1) * cardsPorPagina); html += '
'; for (let l = 0; l < linhas; l++) { html += ''; for (let c = 0; c < colunas; c++) { const idx = l * colunas + c; if (idx < cardsPagina.length) { const card = cardsPagina[idx]; const quarto = card.querySelector('.card-header')?.textContent || ''; const foto = card.querySelector('.card-foto')?.innerHTML || '
Sem Foto
'; const nome = card.querySelector('.card-nome')?.textContent || ''; html += ` `; } else { html += ''; } } html += ''; } html += `
${quarto}
${foto}
${nome}
Página ${p + 1} de ${totalPaginas} | Cards ${inicio} ao ${fim} | Total: ${cards.length}
`; } html += ''; const janela = window.open('', '_blank'); janela.document.write(html); janela.document.close(); janela.onload = function() { setTimeout(() => { janela.focus(); janela.print(); }, 500); }; }
